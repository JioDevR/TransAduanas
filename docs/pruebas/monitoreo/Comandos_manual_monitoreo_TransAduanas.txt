$urls = @(
 "http://127.0.0.1:5173/", "http://127.0.0.1:5174/",
 "http://127.0.0.1:5180/",
 "http://127.0.0.1:5180/api/v1/catalogos/puertos",
 "http://127.0.0.1:8086/api/v1/health",
 "http://127.0.0.1:9086/actuator/prometheus",
 "http://127.0.0.1:9090/-/ready",
 "http://127.0.0.1:3000/api/health",
 "http://127.0.0.1:3100/ready", "http://127.0.0.1:12345/-/ready"
)
foreach ($url in $urls) {
  try { "$url -> $((Invoke-WebRequest -Uri $url `
    -UseBasicParsing -TimeoutSec 10).StatusCode)" }
  catch { "$url -> ERROR: $($_.Exception.Message)" }
}
